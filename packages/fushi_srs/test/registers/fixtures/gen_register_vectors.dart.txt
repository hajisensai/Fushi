bool _p(int oneIn) => _r.nextInt(oneIn) == 0;

String _css(int n, String seed) {
  final StringBuffer b = StringBuffer();
  int i = 0;
  while (b.length < n) {
    b.write('.c$seed$i{color:#${(i * 2654435761 % 0xffffff).toRadixString(16)}} ');
    i++;
  }
  return b.toString();
}

final List<String> _bigCss = <String>[
  _css(600, 'a'),
  _css(512, 'b').substring(0, 512),
  _css(2000, 'ç字'),
];

/// 字段文本：普通文本、内联大样式（会被去重）、已去重的引用行、小样式、`<script>` 里的样式、
/// 畸形 / 嵌套样式、大小写混写的 script 标签。
String _fieldText() {
  final String css = _pick(_bigCss);
  final String body = _pick(<String>['雨', 'snow', '', '\u{1D4E7}x', 'a<b>c</b>']);
  switch (_r.nextInt(12)) {
    case 0:
    case 1:
      return body;
    case 2:
      return '<style>$css</style>$body';
    case 3:
      return srsDedupStyles('<style>$css</style>$body').html;
    case 4:
      return '<style>.small{}</style>$body';
    case 5:
      return '<script>var s="<style>$css</style>";</script>$body';
    case 6:
      return '<SCRIPT type="x">a</sCrIpT >$body<style>$css</style>';
    case 7:
      return '<style>$css<style>$css</style>$body';
    case 8:
      return '<style>${css.substring(0, 511)}</style>$body';
    case 9:
      return '$body<style>$css</style><style>${_pick(_bigCss)}</style>';
    case 10:
      return srsDedupStyles('$body<style>$css</style><style>${_pick(_bigCss)}</style>').html;
    default:
      return 'v${_r.nextInt(4)}';
  }
}

class _Gen {
  _Gen(this.kind, this.entityId) {
    for (final String w in _writers.take(2 + _r.nextInt(_writers.length - 1))) {
      writers.add(SrsEditWriter(writerId: w));
    }
  }

  final SrsEntityKind kind;
  final String entityId;
  final List<SrsEditWriter> writers = <SrsEditWriter>[];
  final List<SrsFieldEdit> made = <SrsFieldEdit>[];
  int clock = _t0 + _r.nextInt(1 << 28);
  bool bad = false;

  /// 随机一段协作：写入、互相同步（应用对方的部分记录 = 并发 / 支配）、删除、恢复。
  void run(int steps, List<String> keys) {
    for (int i = 0; i < steps; i++) {
      final SrsEditWriter w = _pick(writers);
      switch (_r.nextInt(10)) {
        case 0:
        case 1:
          // 同步：应用别人做过的一部分记录（含闭包）。
          if (made.isNotEmpty) {
            for (int k = 0; k < 1 + _r.nextInt(3); k++) {
              w.apply(_pick(made));
            }
          }
        case 2:
          if (made.isNotEmpty && _p(2)) {
            for (final SrsFieldEdit e in made) {
              w.apply(e);
            }
          }
        default:
          if (_r.nextInt(3) != 0) clock += _r.nextInt(100000);
          final String key = _pick(keys);
          final bool b = _p(40) && _castKey(kind, key);
          if (b) bad = true;
          final String hlcWriter = _p(8) ? _pick(_writers) : w.writerId;
          final SrsFieldEdit e = w.create(
            profileUid: 'p',
            entityKind: kind,
            entityId: entityId,
            fieldKey: key,
            value: _valueFor(kind, key, bad: b),
            hlc: Hlc(clock, _p(4) ? _r.nextInt(3) : 0, hlcWriter),
            timeSource: _r.nextBool() ? SrsTimeSource.anchored : SrsTimeSource.wall,
            machine: _p(6),
          );
          made.add(e);
      }
    }
    // 不连续 dot（旧化身重新发布）：ctx 带 cloud 的记录。
    if (_p(4) && made.isNotEmpty) {
      final SrsFieldEdit src = _pick(made);
      made.add(
        SrsFieldEdit(
          profileUid: 'p',
          entityKind: kind,
          entityId: entityId,
          fieldKey: src.fieldKey,
          dot: SrsDot('old#${_r.nextInt(3)}', 5 + _r.nextInt(5)),
          ctx: SrsCausalContext.of(
            vv: <String, int>{src.dot.writerId: src.dot.seq - (_r.nextBool() ? 1 : 0)},
            cloud: <String, Iterable<int>>{
              src.dot.writerId: <int>[src.dot.seq, src.dot.seq + 2],
              'old#9': <int>[3, 4],
            },
          ),
          value: _valueFor(kind, src.fieldKey, bad: false),
          hlc: Hlc(clock + _r.nextInt(3), 0, 'old'),
          timeSource: SrsTimeSource.wall,
        ),
      );
    }
  }
}

/// 落库形态：canonical JSON 原文（偶尔是同 ID 的重复 / 同 ID 异内容）。
List<String> _stored(List<SrsFieldEdit> edits) {
  final List<String> out = <String>[
    for (final SrsFieldEdit e in edits) e.canonical(),
  ];
  if (out.isNotEmpty && _p(6)) out.add(_pick(out));
  if (edits.isNotEmpty && _p(8)) {
    // 同 dot（同 ID）异内容：后到的生效。
    final SrsFieldEdit e = _pick(edits);
    out.add(
      SrsFieldEdit(
        profileUid: e.profileUid,
        entityKind: e.entityKind,
        entityId: e.entityId,
        fieldKey: e.fieldKey,
        dot: e.dot,
        ctx: e.ctx,
        value: 'dup-${_r.nextInt(9)}',
        hlc: e.hlc,
        timeSource: e.timeSource,
      ).canonical(),
    );
  }
  out.shuffle(_r);
  return out;
}

Map<String, int> _serverTimes(List<String> stored) {
  final Map<String, int> s = <String, int>{};
  for (final String j in stored) {
    final Map<String, Object?> m = (jsonDecode(j) as Map).cast<String, Object?>();
    final String id = m['id']! as String;
    final int phys = int.parse((m['hlc']! as String).split(':').first);
    switch (_r.nextInt(5)) {
      case 0:
        s[id] = phys - srsClockToleranceMs - 1 - _r.nextInt(100000);
      case 1:
        s[id] = phys - srsClockToleranceMs;
      case 2:
        s[id] = phys + _r.nextInt(1000);
    }
  }
  return s;
}

List<(String, SrsEntityEdits)> _cardLineage(
  String cardId,
  SrsEntityEdits Function(String id) editsOf,
) {
  final List<(String, SrsEntityEdits)> out = <(String, SrsEntityEdits)>[];
  final Set<String> seen = <String>{cardId};
  final List<String> queue = <String>[cardId];
  while (queue.isNotEmpty && out.length < 256) {
    final String id = queue.removeAt(0);
    final SrsEntityEdits edits = editsOf(id);
    out.add((id, edits));
    final List<String> preds = <String>[
      for (final String k in edits.keys)
        if (k.startsWith(SrsFieldKeys.cardHistoryFromPrefix) &&
            edits.value(k) == true)
          k.substring(SrsFieldKeys.cardHistoryFromPrefix.length),
    ]..sort();
    for (final String p in preds) {
      if (seen.add(p)) queue.add(p);
    }
  }
  return out;
}

SrsEntityEdits _inheritedCardEdits(List<(String, SrsEntityEdits)> lineage) {
  final Map<String, SrsRegister> regs = <String, SrsRegister>{};
  for (final (String _, SrsEntityEdits edits) in lineage.reversed) {
    for (final String k in edits.keys) {
      if (k.startsWith(SrsFieldKeys.cardHistoryFromPrefix)) continue;
      regs[k] = edits[k]!;
    }
  }
  return SrsEntityEdits(regs);
}

/// 实体用例：note / notetype / deck / profile。
Map<String, Object?> _entityCase(SrsEntityKind kind) {
  final String id = '${kind.wire}-${_r.nextInt(1 << 20)}';
  final _Gen g = _Gen(kind, id);
  g.run(1 + _r.nextInt(_p(10) ? 60 : 12), _keysFor(kind));
  final List<String> stored = _stored(g.made);
  final Map<String, int> server = _p(4) ? <String, int>{} : _serverTimes(stored);
  final SrsEntityRecord rec = _record(kind, id);
  final SrsBuiltinNoteType? builtin = kind == SrsEntityKind.noteType ? _builtin() : null;
  final Map<String, Object?> v = <String, Object?>{
    'kind': kind.wire,
    'record': rec.toJson(),
    if (builtin != null) 'builtin': _builtinJson(builtin),
    'edits': stored,
    'server': server,
  };
  try {
    final SrsEntityEdits edits = SrsEntityEdits.build(
      _decode(stored),
      serverTimeOf: (String e) => server[e],
    );
    final Object? view = switch (kind) {
      SrsEntityKind.note => _noteJson(SrsNoteView.materialize(rec, edits)),
      SrsEntityKind.noteType => _noteTypeJson(
        SrsNoteTypeView.materialize(rec, edits, builtin: builtin),
      ),
      SrsEntityKind.deck => _deckJson(SrsDeckView.materialize(rec, edits)),
      _ => null,
    };
    final SrsRegister? del = edits[SrsFieldKeys.deleted];
    v['expect'] = <String, Object?>{
      'regs': _regs(edits),
      'deletedAt': edits.deletedAt,
      'concurrent': del == null
          ? <String>[]
          : <String>[
              for (final SrsFieldEdit e in srsEditsConcurrentWithDelete(del, <SrsFieldEdit>[
                for (final SrsRegister r in edits.registers)
                  if (r.key != SrsFieldKeys.deleted) ...r.live,
              ]))
                e.id,
            ],
      'view': view,
      'noop': _noops(edits),
    };
  } on Object catch (e) {
    v['expect'] = <String, Object?>{'error': '${e.runtimeType}'};
  }
  if (g.bad && v['expect'] is Map && (v['expect']! as Map)['error'] == null) {
    // 坏值没被强转碰到（被别的记录支配 / 不是生效值）：照常记结果。
  }
  return v;
}

List<Object?> _noops(SrsEntityEdits edits) {
  final List<Object?> out = <Object?>[];
  for (final String k in edits.keys) {
    if (!_p(2)) continue;
    final SrsRegister reg = edits[k]!;
    final List<Object?> candidates = <Object?>[
      reg.value,
      if (reg.live.length > 1) reg.live[1].value,
      if (reg.value is String) srsDedupStyles(reg.value! as String).html,
      if (reg.value is String && (reg.value! as String).contains(kSrsStyleMediaPrefix))
        srsExpandStyles(reg.value! as String, (String name) {
          for (final String css in _bigCss) {
            if (srsStyleMediaName(css) == name) return css;
          }
          return null;
        }),
      _anyValue(),
      _fieldText(),
    ];
    final Object? nv = _pick(candidates);
    out.add(<Object?>[k, jsonEncode(nv), srsEditIsNoop(reg, nv)]);
  }
  return out;
}

/// 卡片用例：一张卡 + 前身链（成环、缺失、historyFrom 值不为 true、冲突）。
Map<String, Object?> _cardCase() {
  final List<String> ids = <String>['c0', 'c1', 'c2', '\u{1D4E7}c', 'Ａc', 'c5'];
  final Map<String, List<String>> stored = <String, List<String>>{};
  final Map<String, int> server = <String, int>{};
  bool bad = false;
  for (final String id in ids) {
    if (_p(4) && id != 'c0') continue; // 缺失：没有修改记录的前身
    final _Gen g = _Gen(SrsEntityKind.card, id);
    final List<String> keys = <String>[
      ..._keysFor(SrsEntityKind.card),
      for (final String p in ids)
        if (p != id || _p(10)) '${SrsFieldKeys.cardHistoryFromPrefix}$p',
      '${SrsFieldKeys.cardHistoryFromPrefix}missing',
    ];
    g.run(_r.nextInt(10), keys);
    bad = bad || g.bad;
    stored[id] = _stored(g.made);
    if (!_p(4)) server.addAll(_serverTimes(stored[id]!));
  }
  final String? noteDeck = _pick(<String?>['nd', null]);
  final Map<String, Object?> v = <String, Object?>{
    'kind': 'card',
    'card': 'c0',
    'cards': stored,
    'server': server,
    'noteDeckId': noteDeck,
    'fallbackDeckId': 'fallback',
  };
  try {
    final Map<String, SrsEntityEdits> built = <String, SrsEntityEdits>{
      for (final MapEntry<String, List<String>> e in stored.entries)
        e.key: SrsEntityEdits.build(
          _decode(e.value),
          serverTimeOf: (String x) => server[x],
        ),
    };
    final List<(String, SrsEntityEdits)> lineage = _cardLineage(
      'c0',
      (String id) => built[id] ?? SrsEntityEdits.build(const <SrsFieldEdit>[]),
    );
    final SrsCardFlags flags = SrsCardFlags.materialize(
      _inheritedCardEdits(lineage),
      noteDeckId: noteDeck,
      fallbackDeckId: 'fallback',
    );
    v['expect'] = <String, Object?>{
      'lineage': <String>[for (final (String c, SrsEntityEdits _) in lineage) c],
      'flags': _flagsJson(flags),
      'regs': <String, Object?>{
        for (final MapEntry<String, SrsEntityEdits> e in built.entries) e.key: _regs(e.value),
      },
    };
  } on Object catch (e) {
    v['expect'] = <String, Object?>{'error': '${e.runtimeType}'};
  }
  return v;
}

/// 落库支配裁剪（旧 srs_db_store `_storeLocalBatch` / SrsEventStore `_applyEdit` 的同一规则）：
/// 已有的记录 + 依次到达的记录 → 每条到达的落不落库、哪些已有 / 先到的被支配删掉。
Map<String, Object?> _landingCase() {
  final _Gen g = _Gen(SrsEntityKind.note, 'n');
  g.run(2 + _r.nextInt(14), <String>['f/fa', 'f/fb']);
  final List<SrsFieldEdit> all = <SrsFieldEdit>[...g.made]..shuffle(_r);
  final int split = _r.nextInt(all.length + 1);
  final List<SrsFieldEdit> existing = all.sublist(0, split);
  final List<SrsFieldEdit> incoming = <SrsFieldEdit>[
    ...all.sublist(split),
    if (existing.isNotEmpty && _p(3)) _pick(existing),
    if (all.isNotEmpty && _p(3)) _pick(all),
  ]..shuffle(_r);
  final Map<String, List<SrsFieldEdit>> regs = <String, List<SrsFieldEdit>>{};
  final Set<String> ids = <String>{};
  for (final SrsFieldEdit e in existing) {
    (regs[e.registerKey] ??= <SrsFieldEdit>[]).add(e);
    ids.add(e.id);
  }
  final List<bool> landed = <bool>[];
  final List<String> deleted = <String>[];
  for (final SrsFieldEdit edit in incoming) {
    final List<SrsFieldEdit> reg = regs[edit.registerKey] ??= <SrsFieldEdit>[];
    final bool ok =
        !ids.contains(edit.id) && !reg.any((SrsFieldEdit b) => b.dominates(edit));
    landed.add(ok);
    if (!ok) continue;
    final List<String> dominated = <String>[
      for (final SrsFieldEdit a in reg)
        if (edit.dominates(a)) a.id,
    ];
    reg.removeWhere((SrsFieldEdit a) => dominated.contains(a.id));
    ids.removeAll(dominated);
    deleted.addAll(dominated);
    reg.add(edit);
    ids.add(edit.id);
  }
  return <String, Object?>{
    'kind': 'landing',
    'existing': <String>[for (final SrsFieldEdit e in existing) e.canonical()],
    'incoming': <String>[for (final SrsFieldEdit e in incoming) e.canonical()],
    'expect': <String, Object?>{'landed': landed, 'deleted': deleted},
  };
}

/// 导入判重（旧 srsImportEquivalent，sameCard: true）：一条导入日志对一组原生日志。
Map<String, Object?> _importCase() {
  final int at = _t0 + _r.nextInt(1 << 30);
  SrsReviewLog log(String id, String origin, int occurredAt, int? ankiId) => SrsReviewLog(
    id: id,
    profileUid: 'p',
    cardId: _pick(<String>['c', 'pred']),
    kind: _pick(<String>[SrsLogKind.review, SrsLogKind.review, SrsLogKind.review, SrsLogKind.review, SrsLogKind.manual, 'futureKind']),
    hlc: Hlc(occurredAt, 0, origin == SrsLogOrigin.anki ? 'anki' : 'A'),
    occurredAt: occurredAt,
    timeSource: SrsTimeSource.wall,
    origin: origin,
    reviewType: _pick(<String>[
      SrsReviewType.learn,
      SrsReviewType.review,
      SrsReviewType.relearn,
      SrsReviewType.review,
      SrsReviewType.review,
      SrsReviewType.manual,
      SrsReviewType.filtered,
    ]),
    rating: _p(8) ? null : 1 + _r.nextInt(2),
    ankiRevlogId: ankiId,
  );
  final List<SrsReviewLog> natives = <SrsReviewLog>[
    for (int i = 0; i < _r.nextInt(5); i++)
      log('n$i', _pick(<String>[SrsLogOrigin.fushi, SrsLogOrigin.fushi, SrsLogOrigin.anki]),
          at + _r.nextInt(5000) - 2500, _p(3) ? at : null),
  ];
  final SrsReviewLog? src = natives.isEmpty ? null : _pick(natives);

  final SrsReviewLog raw = log(
    'imp',
    SrsLogOrigin.anki,

    _p(8) ? null : near + _pick(<int>[0, 1, -1, 999, -1000, 1000, 1001, -1001, 2500]),
  );
  // 多数照着一条原生日志镜像（同评分、评分类），等价 / 不等价两侧都要有。
  final SrsReviewLog imported = src != null && !_p(3)
      ? SrsReviewLog(

          cardId: 'c',
          kind: _p(6) ? SrsLogKind.manual : SrsLogKind.review,

  return <String, Object?>{

    'imported': imported.canonical(),
    'natives': <String>[for (final SrsReviewLog l in natives) l.canonical()],

  };
}

/// 卡片历史的钳位标记（旧 SrsCardHistory.sortKeyOf：钳位后的物理时间 < HLC 物理时间）。
Map<String, Object?> _clampCase() {
  final List<SrsReviewLog> logs = <SrsReviewLog>[];
  final Map<String, int> server = <String, int>{};
  int clock = _t0 + _r.nextInt(1 << 28);
  for (int i = 0; i < 1 + _r.nextInt(8); i++) {
    clock += _r.nextInt(3) == 0 ? 0 : _r.nextInt(10000000);
    final String id = 'L$i';
    logs.add(
      SrsReviewLog(
        id: id,
        profileUid: 'p',
        cardId: 'c',
        kind: SrsLogKind.review,
        hlc: Hlc(clock, 0, _pick(_writers)),
        occurredAt: clock,
        timeSource: SrsTimeSource.wall,
        origin: SrsLogOrigin.fushi,
        reviewType: SrsReviewType.review,
        rating: 3,
        result: SrsScheduleResult.initial,
      ),
    );
    switch (_r.nextInt(4)) {
      case 0:
        server[id] = clock - srsClockToleranceMs - 1 - _r.nextInt(1000000);
      case 1:
        server[id] = clock - srsClockToleranceMs;
      case 2:
        server[id] = clock + 5;
    }
  }
  final SrsCardHistory h = SrsCardHistory(logs, serverTimeOf: (String e) => server[e]);
  return <String, Object?>{
    'kind': 'clamp',
    'logs': <String>[for (final SrsReviewLog l in logs) l.canonical()],
    'server': server,
    'expect': <String, Object?>{
      'all': <String>[for (final SrsReviewLog l in h.all) l.id],
      'clamped': <String>[
        for (final SrsReviewLog l in h.all)
          if (h.sortKeyOf(l).physical < l.hlc.physical) l.id,
      ],
    },
  };
}
