  final Random r = Random(seed);
  return List<String>.generate(
    40,
    (_) => '0123456789abcdef'[r.nextInt(16)],
  ).join();
}

/// 语料随机拼接（字段值 / 纯函数输入）。
String randomValue(List<String> corpus) {
  final int roll = _r.nextInt(10);
  if (roll < 5) return _pick(corpus);
  final int n = 2 + _r.nextInt(3);
  return List<String>.generate(
    n,
    (_) => _pick(corpus),
  ).join(_pick(<String>['', ' ', '\n', '<br>', '・']));
}

bool _hasLoneSurrogate(String s) {
  for (int i = 0; i < s.length; i++) {
    final int c = s.codeUnitAt(i);
    if (c >= 0xD800 && c <= 0xDBFF) {
      if (i + 1 < s.length) {
        final int d = s.codeUnitAt(i + 1);
        if (d >= 0xDC00 && d <= 0xDFFF) {
          i++;
          continue;
        }
      }
      return true;
    }
    if (c >= 0xDC00 && c <= 0xDFFF) return true;
  }
  return false;
}

String _sortOf(String s) {
  final String t = ankiStripHtml(s).trim();
  return t.length > 200 ? t.substring(0, 200) : t;
}

/// 同 srs_db_store.dart `_mediaRefsOf`（单个值）。
String _mediaOf(String s) {
  final Set<String> refs = <String>{...srsMediaRefsIn(s), ...srsStyleRefsIn(s)};
  return (refs.where((String r) => !r.contains('\n')).toList()..sort()).join(
    '\n',
  );
}

Object? _jsonCell(Object? v) =>
    v is Uint8List ? <String, String>{'b64': base64Encode(v)} : v;

/// 一张表全部列的全部行（按主键排序）。
Future<Map<String, Object?>> dumpTable(
  SrsDbStore s,
  String table, {
  Set<String> drop = const <String>{},
}) async {
  final List<QueryRow> info = await s.db
      .customSelect('PRAGMA table_info($table)')
      .get();
  final List<String> cols = <String>[
    for (final QueryRow c in info)
      if (!drop.contains(c.read<String>('name'))) c.read<String>('name'),
  ];
  final List<QueryRow> pkRows =
      info.where((QueryRow c) => c.read<int>('pk') > 0).toList()..sort(
        (QueryRow a, QueryRow b) =>
            a.read<int>('pk').compareTo(b.read<int>('pk')),
      );
  final List<String> pk = <String>[
    for (final QueryRow c in pkRows) c.read<String>('name'),
  ];
  final List<QueryRow> rows = await s.db
      .customSelect(
        'SELECT ${cols.map((String c) => '"$c"').join(', ')} FROM $table '
        'WHERE profile_uid = ? ORDER BY ${pk.map((String c) => '"$c"').join(', ')}',
        variables: <Variable<Object>>[Variable<String>(s.profileUid)],
      )
      .get();
  return <String, Object?>{
    'columns': cols,
    'pk': pk,
    'rows': <List<Object?>>[
      for (final QueryRow r in rows)
        <Object?>[for (final String c in cols) _jsonCell(r.data[c])],
    ],
  };
}

    }

    final SrsDbStore a = await device(kDevice);
    await a.editFields(SrsEntityKind.profile, kProfile, <String, Object?>{
      SrsFieldKeys.utcOffsetMin: 540,
      SrsFieldKeys.dayRolloverHour: 4,
    });

    // ---------------------------------------------------------------- 牌组
    final String deckSub = await a.ensureDeckPath('Vec::Sub');
    final String deckLeaf = await a.ensureDeckPath('Vec::Sub::Leaf');
    final String deck2 = await a.createDeck('Vec2');
    final List<String?> decks = <String?>[

    // ---------------------------------------------------------------- 笔记类型

      for (int i = 0; i < count; i++) {

    tick();
    await a.setTermField(typeC, (await type(typeC)).fields.first.id);

    // 更改笔记类型：D → A（含复习过的卡），再把其中一部分 A → C（两层血统）；Lapis → D。
    final List<String> dNotes = <String>[
      for (final String n in notes)
        if (noteType[n] == typeD) n,
    ];
    final List<String> toA = dNotes.take(14).toList();
    va = await type(typeA);
    final SrsNoteTypeView vdNow = await type(typeD);
    tick();
    await a.changeNoteType(
      noteIds: toA,
      targetTypeId: typeA,
      fieldMap: srsDefaultFieldMap(vdNow, va),
      templateMap: srsDefaultTemplateMap(vdNow, va),
    );
    for (final String n in toA) {
      noteType[n] = typeA;
    }
    await reviewRound(40);
    final List<String> toC = toA.take(6).toList();
    final SrsNoteTypeView vc = await type(typeC);
    va = await type(typeA);
    tick();
    await a.changeNoteType(
      noteIds: toC,
      targetTypeId: typeC,
      fieldMap: srsDefaultFieldMap(va, vc),
      templateMap: srsDefaultTemplateMap(va, vc),
    );
    for (final String n in toC) {
      noteType[n] = typeC;
    }
    final List<String> lapisNotes = <String>[
      for (final String n in notes)
        if (noteType[n] == a.lapisNoteTypeId) n,
    ].take(8).toList();
    final SrsNoteTypeView vl = await type(a.lapisNoteTypeId);
    final SrsNoteTypeView vd2 = await type(typeD);
    tick();
    await a.changeNoteType(
      noteIds: lapisNotes,
      targetTypeId: typeD,
      fieldMap: srsDefaultFieldMap(vl, vd2)
        ..[vd2.fields.first.id] = vl.fields.first.id,
      templateMap: srsDefaultTemplateMap(vl, vd2),
    );
    for (final String n in lapisNotes) {
      noteType[n] = typeD;
    }
    await reviewRound(40);

    // 卡片标志。
    final List<SrsCardCacheRow> all = await a.allCardRows();
    final List<SrsCardCacheRow> shown = all
        .where((SrsCardCacheRow c) => !c.hidden)
        .toList();
    for (int i = 0; i < 30; i++) {
      tick();
      await a.setSuspended(_pick(shown).id, _r.nextInt(4) != 0);
    }
    for (int i = 0; i < 20; i++) {
      tick();
      await a.buryUntil(
        _pick(shown).id,
        _r.nextInt(4) == 0 ? null : 20400 + _r.nextInt(300),
      );
    }
    for (int i = 0; i < 30; i++) {
      tick();
      await a.setFlag(_pick(shown).id, _r.nextInt(8));
    }
    for (int i = 0; i < 30; i++) {
      tick();
      await a.moveCard(
        _pick(all).id,
        _pick(<String>[deckSub, deckLeaf, deck2, a.defaultDeckId]),
      );
    }
    final List<SrsCardCacheRow> fresh =
        shown.where((SrsCardCacheRow c) => c.queue == 'new').toList()
          ..shuffle(_r);
    tick();
    await a.setNewCardPositions(<String, int>{
      for (final SrsCardCacheRow c in fresh.take(40))
        c.id: _r.nextInt(500) - 50,
    });

    // 删除 / 恢复笔记、删类型。
    final List<String> shuffled = notes.toList()..shuffle(_r);
    final List<String> deleted = shuffled.take(22).toList();
    for (final String n in deleted) {
      tick();
      await a.deleteNote(n);
    }
    for (final String n in deleted.take(5)) {
      tick();
      await a.restoreNote(n);
    }
    tick();
    await a.deleteNoteType(typeDel);

    // ---------------------------------------------------------------- 第二台设备
    final SrsMemoryLocationStore loc = SrsMemoryLocationStore();
    final SrsDbStore b = await device('vec-B');
    await srsSyncPeers(<SrsDbStore>[a, b], loc, rounds: 2);
    // 删除 vs 并发修改 → 待裁决；B 端另有修改与复习。
    final List<String> live = <String>[
      for (final String n in shuffled.skip(22)) n,
    ];
    for (final String n in live.take(6)) {
      tick();
      await a.deleteNote(n);
      final SrsNoteTypeView v = (await b.noteType(noteType[n]!))!;
      await b.updateNote(
        n,
        fieldsById: <String, String>{v.fields.first.id: 'B改${_r.nextInt(100)}'},
      );
    }
    for (final String n in live.skip(6).take(20)) {
      tick();
      final SrsNoteTypeView v = (await b.noteType(noteType[n]!))!;
      await b.updateNote(
        n,
        fieldsById: <String, String>{_pick(v.fields).id: randomValue(corpus)},
        tags: <String>['from-b'],
      );
    }
    final List<SrsCardCacheRow> bShown = (await b.allCardRows())
        .where((SrsCardCacheRow c) => !c.hidden)
        .toList();
    for (int i = 0; i < 40; i++) {
      t += _pick(<int>[600000, 86400000, 4 * 86400000]);
      await b.answer(_pick(bShown).id, 1 + _r.nextInt(4));
    }
    for (int i = 0; i < 10; i++) {
      tick();
      await b.setFlag(_pick(bShown).id, 1 + _r.nextInt(6));
    }
    await srsSyncPeers(<SrsDbStore>[a, b], loc, rounds: 2);
    await reviewRound(30);

    // ---------------------------------------------------------------- 不变量：增量 == 重建
    final Map<String, Map<String, Object?>> incremental = await dumpCaches(a);
    await a.rebuildCaches();
    final Map<String, Map<String, Object?>> rebuilt = await dumpCaches(a);
    final Map<String, List<String>> succ = await a.successorIndex();
    final List<String> brokenTables = <String>[];
    for (final String table in incremental.keys) {
      final List<String> cols =
          (incremental[table]!['columns']! as List<Object?>).cast<String>();
      Map<Object?, List<Object?>> byKey(List<Object?> rows) =>
          <Object?, List<Object?>>{
            for (final Object? r in rows)
              (r! as List<Object?>).first: r as List<Object?>,
          };
      final Map<Object?, List<Object?>> x = byKey(
        incremental[table]!['rows']! as List<Object?>,
      );
      final Map<Object?, List<Object?>> y = byKey(
        rebuilt[table]!['rows']! as List<Object?>,
      );
      final List<Object?> onlyX = <Object?>[
        for (final Object? k in x.keys)
          if (!y.containsKey(k)) k,
      ];
      final List<Object?> onlyY = <Object?>[
        for (final Object? k in y.keys)
          if (!x.containsKey(k)) k,
      ];
      final List<Object?> differ = <Object?>[
        for (final Object? k in x.keys)
          if (y.containsKey(k) && jsonEncode(x[k]) != jsonEncode(y[k])) k,
      ];
      stdout.writeln(
        'INVARIANT $table: incremental=${x.length} rebuild=${y.length} '
        'onlyIncremental=${onlyX.length} onlyRebuild=${onlyY.length} differ=${differ.length}',
      );
      // 放宽后的不变量（协调方拍板 A）：增量多出的行只能是 hidden=1 的「不再应有的卡」（重建不写它们）；
      // 重建不能多行，共有的行逐列一致。期望表取重建结果。
      final int hiddenCol = cols.indexOf('hidden');
      final bool extrasHidden =
          hiddenCol >= 0 && onlyX.every((Object? k) => x[k]![hiddenCol] == 1);
      if (onlyY.isEmpty && differ.isEmpty && (onlyX.isEmpty || extrasHidden)) {
        if (onlyX.isEmpty) continue;
        stdout.writeln(
          '  (allowed: ${onlyX.length} hidden-only extra rows in incremental)',
        );
      } else {
        brokenTables.add(table);
      }
      if (table == 'srs_cards_cache') {
        final int hi = cols.indexOf('hidden');
        final int tpl = cols.indexOf('template_id');
        final int ord = cols.indexOf('cloze_ord');
        final int reps = cols.indexOf('reps');
        final Map<String, int> why = <String, int>{};
        for (final Object? k in onlyX) {
          final List<Object?> r = x[k]!;
          final String c =
              '${r[hi] == 1 ? 'hidden' : 'VISIBLE'}/'
              '${succ.containsKey(k) ? 'predecessor' : 'no-successor'}/'
              '${(r[ord]! as int) > 0 ? 'cloze' : 'tpl'}/${(r[reps]! as int) > 0 ? 'reviewed' : 'unreviewed'}';
          why[c] = (why[c] ?? 0) + 1;
        }
        stdout.writeln('  onlyIncremental by hidden/lineage/kind/reps: $why');
        final Map<Object?, int> byTpl = <Object?, int>{};
        for (final Object? k in onlyX) {
          byTpl[x[k]![tpl]] = (byTpl[x[k]![tpl]] ?? 0) + 1;
        }
        stdout.writeln('  onlyIncremental template ids: $byTpl');
      }
      for (final Object? k in onlyX.take(3)) {
        stdout.writeln('  incremental only: ${jsonEncode(x[k])}');
      }
      for (final Object? k in onlyY.take(3)) {
        stdout.writeln('  rebuild only:     ${jsonEncode(y[k])}');
      }
      for (final Object? k in differ.take(5)) {
        stdout.writeln('  differ $k:');
        for (int i = 0; i < cols.length; i++) {
          if (jsonEncode(x[k]![i]) != jsonEncode(y[k]![i])) {
            stdout.writeln(
              '    ${cols[i]}: incremental=${jsonEncode(x[k]![i])} rebuild=${jsonEncode(y[k]![i])}',
            );
          }
        }
      }
    }
    if (brokenTables.isNotEmpty)
      fail('cache invariant broken in $brokenTables (vectors not written)');
    stdout.writeln('INVARIANT ok (relaxed: hidden-only extras allowed)');

    // ---------------------------------------------------------------- 导出 store 级向量
    final Map<String, Object?> raw = <String, Object?>{
      for (final String table in kRawTables) table: await dumpTable(a, table),
    };
    final Map<String, Object?> store = <String, Object?>{
      'profile': kProfile,
      'device': kDevice,
      'tables': raw,
      'expected': rebuilt,
    };
    final File gz = File('test/srs/fixtures/cache_vectors.json.gz')
      ..parent.createSync(recursive: true);
    gz.writeAsBytesSync(gzip.encode(utf8.encode(jsonEncode(store))));
    final List<Object?> nc =
        rebuilt['srs_notes_cache']!['rows']! as List<Object?>;
    final List<Object?> cc =
        rebuilt['srs_cards_cache']!['rows']! as List<Object?>;
    final List<Object?> sr =
        rebuilt['srs_stat_reviews']!['rows']! as List<Object?>;
    final List<String> ncCols =
        (rebuilt['srs_notes_cache']!['columns']! as List<Object?>)
            .cast<String>();
    final List<String> ccCols =
        (rebuilt['srs_cards_cache']!['columns']! as List<Object?>)
            .cast<String>();
    int count(
      List<Object?> rows,
      List<String> cols,
      String col,
      bool Function(Object?) p,
    ) => rows
        .where((Object? r) => p((r! as List<Object?>)[cols.indexOf(col)]))
        .length;
    stdout.writeln(
      'STORE notes=${nc.length} cards=${cc.length} stat=${sr.length} '
      'logs=${((raw['srs_review_logs']! as Map<String, Object?>)['rows']! as List<Object?>).length} '
      'entities=${((raw['srs_entities']! as Map<String, Object?>)['rows']! as List<Object?>).length} '
      'edits=${((raw['srs_field_edits']! as Map<String, Object?>)['rows']! as List<Object?>).length} '
      'eventLocal=${((raw['srs_event_local']! as Map<String, Object?>)['rows']! as List<Object?>).length}',
    );
    stdout.writeln(
      'STORE hiddenCards=${count(cc, ccCols, 'hidden', (Object? v) => v == 1)} '
      'deletedNotes=${count(nc, ncCols, 'deleted_at', (Object? v) => v != null)} '
      'pending=${count(nc, ncCols, 'pending_decision', (Object? v) => v == 1)} '
      'withReading=${count(nc, ncCols, 'reading_key', (Object? v) => v != '')} '
      'withMedia=${count(nc, ncCols, 'media_refs', (Object? v) => v != '')} '
      'sortHasFFFD=${count(nc, ncCols, 'sort_field', (Object? v) => (v! as String).contains('�'))} '
      'suspended=${count(cc, ccCols, 'suspended', (Object? v) => v == 1)} '
      'flagged=${count(cc, ccCols, 'flag', (Object? v) => v != 0)} '
      'newPos=${count(cc, ccCols, 'new_position', (Object? v) => v != null)} '
      'clozeOrd>0=${count(cc, ccCols, 'cloze_ord', (Object? v) => v != 0)} '
      'reviewed=${count(cc, ccCols, 'reps', (Object? v) => v != 0)}',
    );
    stdout.writeln('wrote ${gz.path} (${gz.lengthSync()} bytes)');

    // ---------------------------------------------------------------- 纯函数向量
    final List<Map<String, Object?>> text = <Map<String, Object?>>[];
    final List<String> mediaThrows = <String>[];
    int skippedSurrogate = 0;
    final List<String> inputs = <String>[
      ...corpus,
      ...throwing,
      for (final String tpl in kTemplates) tpl,
      for (int i = 0; i < 2400; i++)
        randomValue(<String>[...corpus, ...throwing]),
      for (int i = 0; i < 1600; i++)
        List<String>.generate(
          2 + _r.nextInt(3),
          (_) => _pick(<String>[...corpus, ...throwing]),
        ).join(_pick(<String>['', ' ', '\n'])),
    ];
    final Set<String> seen = <String>{};
    for (final String s in inputs) {
      if (!seen.add(s)) continue;
      // media = null：旧实现在这里抛（Uri.decodeFull 的 FormatException，`_unescape` 只接 ArgumentError）；
      // Rust 版有意不抛、原样保留，测试跳过 null。
      String? media;
      try {
        media = _mediaOf(s);
      } on Object {
        mediaThrows.add(s);
      }
      final String term = srsNormalizeTerm(s);
      final String sort = _sortOf(s);
      final List<int> clozeOrds = clozeOrdinalsIn(s).toList();
      if (<String>[s, term, sort, ?media].any(_hasLoneSurrogate)) {
        skippedSurrogate++;
        continue;
      }
      text.add(<String, Object?>{
        's': s,
        'term': term,
        'sort': sort,
        'cloze': clozeOrds,
        'media': media,
      });
    }

    // 模板判空。
    const Map<String, String> allIds = <String, String>{
      'Front': 'fF',
      'Back': 'fB',
      'Extra': 'fE',
      'Reading': 'fR',
    };
    const List<String> emptyish = <String>[
      '',
      ' ',
      '<br>',
      '<div></div>',
      '<div><br></div>',
      '&nbsp;',
      '　',
      '\n',
      'x',
      '0',
      '<img src="a.png">',
      '<b></b>',
      '猫',
      '{{c1::x}}',
      '\u0085',
      '﻿',
      '<!-- c -->',
    ];
    final List<Map<String, Object?>> empties = <Map<String, Object?>>[];
    int nTrue = 0, nFalse = 0, nNull = 0;
    for (final String named in <String>[
      ...kTemplates,
      '{{fld:ghost}}',
      '{{#fld:ghost}}x{{/fld:ghost}}',
    ]) {
      for (int k = 0; k < 30; k++) {
        // 偶尔 idToName 缺一个字段（模板里的 fld:id 原样留下）。
        final Map<String, String> nameToId = Map<String, String>.of(allIds);
        if (k % 7 == 6) nameToId.remove(_pick(allIds.keys.toList()));
        final String front = templateNamesToIds(named, allIds);
        final Map<String, String> idToName = <String, String>{
          for (final MapEntry<String, String> e in nameToId.entries)
            e.value: e.key,
        };
        final Map<String, String> fields = <String, String>{
          for (final String name in nameToId.keys)
            name: _r.nextInt(4) == 0 ? randomValue(corpus) : _pick(emptyish),
        };
        final int ord = _r.nextInt(4);
        bool? empty;
        try {
          empty = isAnkiCardEmpty(
            questionTemplate: templateIdsToNames(front, idToName),
            noteFields: fields,
            cardOrd: ord,
            isCloze: false,
          );
        } on Object {
          empty = null;
        }
        if (<String>[front, ...fields.values].any(_hasLoneSurrogate)) continue;
        if (empty == null) {
          nNull++;
        } else if (empty) {
          nTrue++;
        } else {
          nFalse++;
        }
        empties.add(<String, Object?>{
          'front': front,
          'idToName': idToName,
          'fields': fields,
          'ord': ord,
          'empty': empty,
        });
      }
    }
    final File tf = File(
      '../../native/fushi_fsrs/tests/fixtures/cache_text_vectors.json',
    )..parent.createSync(recursive: true);
    tf.writeAsStringSync(
      jsonEncode(<String, Object?>{
        'source':
            'packages/fushi_engine/test/srs/fixtures/gen_cache_vectors.dart.txt (Dart implementation before the Rust port)',
        'text': text,
        'templateEmpty': empties,
      }),
    );
    stdout.writeln(
      'TEXT entries=${text.length} mediaNull=${mediaThrows.length} skippedSurrogate=$skippedSurrogate '
      'templateEmpty=${empties.length} (true=$nTrue false=$nFalse null=$nNull) -> ${tf.path} (${tf.lengthSync()} bytes)',
    );
    expect(tf.lengthSync(), lessThan(2 * 1024 * 1024));
  }, timeout: const Timeout(Duration(minutes: 40)));
}
